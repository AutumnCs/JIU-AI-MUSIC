#### 伴学小鸟悬浮组件 (BirdCompanion)

```tsx
'use client';
import { useState } from 'react';
import { useBirdStore } from '@/stores/birdStore';
import { motion, AnimatePresence } from 'framer-motion';

export function BirdCompanion() {
  const { currentBird } = useBirdStore();
  const [isChatOpen, setIsChatOpen] = useState(false);

  return (
    <div className="fixed bottom-20 right-4 z-50">
      <AnimatePresence>
        {isChatOpen && (
          <motion.div
            initial={{ opacity: 0, y: 20 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: 20 }}
            className="mb-4 w-72 rounded-2xl bg-white shadow-xl"
          >
            <ChatPanel bird={currentBird} />
          </motion.div>
        )}
      </AnimatePresence>
      <motion.button
        whileHover={{ scale: 1.1 }}
        whileTap={{ scale: 0.9 }}
        onClick={() => setIsChatOpen(!isChatOpen)}
        className="h-16 w-16 rounded-full shadow-lg"
      >
        <img src={currentBird.avatar} alt={currentBird.name} />
      </motion.button>
    </div>
  );
}
```

#### 音高检测 Hook

```tsx
'use client';
import { useRef, useCallback, useState } from 'react';
import { PitchDetector } from 'pitchy';

export function usePitchDetection() {
  const [pitch, setPitch] = useState<number | null>(null);
  const [note, setNote] = useState<string | null>(null);
  const detectorRef = useRef<PitchDetector<Float32Array> | null>(null);
  const animFrameRef = useRef<number>(0);

  const startListening = useCallback(async () => {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    const audioContext = new AudioContext();
    const source = audioContext.createMediaStreamSource(stream);
    const analyzer = audioContext.createAnalyser();
    analyzer.fftSize = 2048;
    source.connect(analyzer);

    detectorRef.current = PitchDetector.forFloat32Array(2048);
    const data = new Float32Array(2048);

    const detect = () => {
      analyzer.getFloatTimeDomainData(data);
      const [freq, clarity] = detectorRef.current!.findPitch(
        data, audioContext.sampleRate
      );
      if (clarity > 0.8 && freq > 0) {
        setPitch(freq);
        setNote(hzToNoteName(freq));
      }
      animFrameRef.current = requestAnimationFrame(detect);
    };
    detect();
  }, []);

  const stopListening = useCallback(() => {
    cancelAnimationFrame(animFrameRef.current);
    setPitch(null);
    setNote(null);
  }, []);

  return { pitch, note, startListening, stopListening };
}

function hzToNoteName(hz: number): string {
  const notes = ['C','C#','D','D#','E','F','F#','G','G#','A','A#','B'];
  const semitones = 12 * Math.log2(hz / 440);
  const noteIndex = Math.round(semitones) + 9;
  const octave = Math.floor(noteIndex / 12) + 4;
  return `${notes[noteIndex % 12]}${octave}`;
}
```

### 3.2 核心 API 实现示例

```python
# routers/workshop.py
from fastapi import APIRouter, UploadFile, File, Form
from services.music_generator import MusicGenerator
from services.melody_analyzer import MelodyAnalyzer

router = APIRouter(prefix="/api/workshop", tags=["workshop"])
music_gen = MusicGenerator()
melody_analyzer = MelodyAnalyzer()

@router.post("/generate")
async def generate_song(
    humming: UploadFile = File(...),
    style: str = Form("欢快"),
    ambient: UploadFile = File(None),
):
    humming_path = await save_upload(humming)
    melody_info = melody_analyzer.analyze(humming_path)
    template = select_template(style, melody_info["key"])
    
    if ambient:
        ambient_path = await save_upload(ambient)
        result = await music_gen.generate_with_ambient(
            melody=melody_info, template=template,
            ambient=ambient_path, style=style,
        )
    else:
        result = await music_gen.generate(
            melody=melody_info, template=template, style=style,
        )
    
    return {
        "audio_url": result["url"],
        "duration": result["duration"],
        "style": style,
        "melody_key": melody_info["key"],
    }
```

### 3.3 AI 音乐生成方案（双轨策略）

```
方案 A（实时生成，质量不可控）：
  哼唱 → librosa 提取旋律 → ACE-Step 生成 → 输出
  优点：真正的 AI 生成
  缺点：质量不稳定，生成时间长

方案 B（预生成 + 智能匹配，质量可控）：
  预先生成 9 首高质量歌曲（3 风格 × 3 调式）
  哼唱 → 分析调式和风格 → 匹配最接近的预生成歌曲
  优点：质量高，速度快
  缺点：不是真正的实时生成

推荐：Demo 阶段用方案 B 保底 + 方案 A 作为增强
```

## 五、关键技术决策记录

| 决策 | 选择 | 理由 |
|------|------|------|
| 前端框架 | Next.js + React | SSR/SSG 支持、生态丰富、部署简单 |
| 音频处理 | Tone.js | 浏览器端最成熟的 Web Audio 框架 |
| 音高检测 | pitchy (npm) | 轻量、纯 JS、无需 WASM |
| 后端框架 | FastAPI | Python 生态（librosa/ACE-Step）、异步支持 |
| AI 音乐生成 | ACE-Step + 预生成 | 双轨策略平衡质量和稳定性 |
| 状态管理 | Zustand | 轻量、简单、TypeScript 友好 |
| 拖拽 | @dnd-kit | React 生态最活跃的拖拽库 |
| 样式 | TailwindCSS | 快速开发、设计稿还原度高 |
| 部署 | Vercel + 阿里云 | 前端零配置部署，后端灵活可控 |
